     Log.e(\"BackendWS\", \"Registration error\", e);\n                }\n            }\n            \n            @Override\n            public void onMessage(WebSocket webSocket, String text) {\n                try {\n                    JSONObject message = new JSONObject(text);\n                    String type = message.getString(\"type\");\n                    \n                    switch       \n                        case \"ADVISORY\":\n                            handleAdvisory(message.getJSONObject(\"payload\"));\n                            break;\n                            \n                        case \"ERROR\":\n                            Log.e(\"BackendWS\", \"Backend error: \" + message.getString(\"message\"));\n                            break;\n                    }\n                } catch (JSONException e) {\n                    Log.e(\"BackendWS\", \"Message parsing error\", e);\n                }\n            }\n            \n            @Override\n            public void onFailure(WebSocket webSocket, Throwable t, Response response) {\n                Log.e(\"BackendWS\", \"Connection failed\", t);\n                isConnected = false;\n                // Implement reconnection logic\n                scheduleReconnect();\n            }\n        });\n    }\n    \n    private void handleAdvisory(JSONObject advisory) {\n        try {\n            boolean validationRequired = advisory.getBoolean(\"validationRequired\");\n            boolean rfRequired = advisory.getBoolean(\"rfDetectionRequired\");\n            boolean biometricRequired = advisory.getBoolean(\"biometricCheckRequired\");\n            String riskLevel = advisory.getString(\"riskLevel\");\n            String journeyId = advisory.getString(\"journeyId\");\n            \n            Log.d(\"BackendWS\", String.format(\n                \"Advisory received: validation=%b, rf=%b, biometric=%b, risk=%s\",\n                validationRequired, rfRequired, biometricRequired, riskLevel\n            ));\n            \n            // Apply advisory to validation controller\n            validationController.applyAdvisory(advisory);\n            \n            // Send   ack.put(\"journeyId\", journeyId);\n            ack.put(\"timestamp\", System.currentTimeMillis());\n            \n            webSocket.send(ack.toString());\n            Log.d(\"BackendWS\", result.put(\"method\", method); // BLE | BIOMETRIC | NFC\n            result.put(\"timestamp\", System.currentTimeMillis());\n            \n            webSocket.send(result.toString());\n            Log.d(\"BackendWS\", String.format(\n                \"Validation result sent: %s via %s\", status, method\n            ));\n        } catch (JSONException e) {\n            Log.e(\"BackendWS\",
